## Solution

```python
class Solution:
    def nthUglyNumber(self, n: int) -> int:
        ugly = [1] * n

        i2 = i3 = i5 = 0

        for i in range(1, n):
            n2 = ugly[i2] * 2
            n3 = ugly[i3] * 3
            n5 = ugly[i5] * 5

            nxt = min(n2, n3, n5)
            ugly[i] = nxt

            if nxt == n2:
                i2 += 1
            if nxt == n3:
                i3 += 1
            if nxt == n5:
                i5 += 1

        return ugly[n - 1]
```
