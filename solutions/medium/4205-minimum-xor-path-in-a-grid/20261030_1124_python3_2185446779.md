## Solution

```python
class Solution:
    def minCost(self, grid: list[list[int]]) -> int:
        m, n = len(grid), len(grid[0])

        reachable = [[set() for _ in range(n)] for _ in range(m)]
        reachable[0][0] = {grid[0][0]}

        for i in range(m):
            for j in range(n):
                if i == 0 and j == 0:
                    continue
                cur = set()
                if i > 0:
                    cur |= reachable[i - 1][j]
                if j > 0:
                    cur |= reachable[i][j - 1]
                reachable[i][j] = {x ^ grid[i][j] for x in cur}

        return min(reachable[m - 1][n - 1])
```
