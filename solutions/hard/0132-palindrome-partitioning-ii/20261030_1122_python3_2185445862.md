## Solution

```python
class Solution:
    def minCut(self, s: str) -> int:
        n = len(s)
        # is_pal[i][j] = s[i..j] is a palindrome
        is_pal = [[False] * n for _ in range(n)]
        for i in range(n - 1, -1, -1):
            for j in range(i, n):
                if s[i] == s[j] and (j - i <= 2 or is_pal[i + 1][j - 1]):
                    is_pal[i][j] = True

        # dp[i] = min cuts for prefix s[0..i-1]
        dp = [0] * (n + 1)
        for i in range(1, n + 1):
            dp[i] = i - 1  # worst case: cut every char
            for j in range(i):
                if is_pal[j][i - 1]:
                    dp[i] = min(dp[i], dp[j] + (0 if j == 0 else 1))

        return dp[n]
```
