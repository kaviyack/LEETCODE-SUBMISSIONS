from typing import List

class Solution:
    def palindromePath(self, n: int, edges: List[List[int]], s: str,
                       queries: List[str]) -> List[bool]:
        # ---------- build tree ----------
        adj = [[] for _ in range(n)]
        for u, v in edges:
            adj[u].append(v)
            adj[v].append(u)

        parent = [-1] * n
        depth = [0] * n
        size = [1] * n
        heavy = [-1] * n

        # iterative DFS for parent/depth/order
        order = []
        st = [0]
        parent[0] = 0
        while st:
            u = st.pop()
            order.append(u)
            for w in adj[u]:
                if w != parent[u]:
                    parent[w] = u
                    depth[w] = depth[u] + 1
                    st.append(w)

        # compute subtree sizes and heavy child (reverse order)
        for u in reversed(order):
            maxsz = 0
            for w in adj[u]:
                if w != parent[u]:
                    size[u] += size[w]
                    if size[w] > maxsz:
                        maxsz = size[w]
                        heavy[u] = w

        # ---------- HLD positions ----------
        head = [0] * n
        pos = [0] * n
        base = [0] * n          # segment-tree base value (char bit at pos)
        cur = 0
        # iterative decomposition
        stack = [(0, 0)]        # (start node, chain head)
        while stack:
            start, h = stack.pop()
            u = start
            while u != -1:
                head[u] = h
                pos[u] = cur
                base[cur] = 1 << (ord(s[u]) - 97)
                cur += 1
                for w in adj[u]:
                    if w != parent[u] and w != heavy[u]:
                        stack.append((w, w))
                u = heavy[u]

        # ---------- segment tree (range XOR, point update) ----------
        N = 1
        while N < n:
            N <<= 1
        seg = [0] * (2 * N)
        for i in range(n):
            seg[N + i] = base[i]
        for i in range(N - 1, 0, -1):
            seg[i] = seg[i << 1] ^ seg[i << 1 | 1]

        def update(i, val):
            p = i + N
            seg[p] = val
            p >>= 1
            while p:
                seg[p] = seg[p << 1] ^ seg[p << 1 | 1]
                p >>= 1

        def query(lo, hi):   # inclusive
            res = 0
            lo += N
            hi += N + 1
            while lo < hi:
                if lo & 1:
                    res ^= seg[lo]
                    lo += 1
                if hi & 1:
                    hi -= 1
                    res ^= seg[hi]
                lo >>= 1
                hi >>= 1
            return res

        def path_xor(u, v):
            res = 0
            while head[u] != head[v]:
                if depth[head[u]] < depth[head[v]]:
                    u, v = v, u
                res ^= query(pos[head[u]], pos[u])
                u = parent[head[u]]
            if depth[u] > depth[v]:
                u, v = v, u
            res ^= query(pos[u], pos[v])
            return res

        # ---------- process queries ----------
        ans = []
        for q in queries:
            parts = q.split()
            if parts[0] == "update":
                node = int(parts[1])
                c = parts[2]
                update(pos[node], 1 << (ord(c) - 97))
            else:
                u = int(parts[1])
                v = int(parts[2])
                mask = path_xor(u, v)   # includes lca bit naturally since range is inclusive
                ans.append((mask & (mask - 1)) == 0)
        return ans
