## Solution

```python
import bisect
import math
from sortedcontainers import SortedList  # Or use bisect with a list


class Solution:

  def maxSumSubmatrix(self, matrix: list[list[int]], k: int) -> int:
    m, n = len(matrix), len(matrix[0])
    max_sum = -math.inf

    for r1 in range(m):
      # col_sums will store the sum of elements from row r1 to r2 for each column
      col_sums = [0] * n
      for r2 in range(r1, m):
        for c in range(n):
          col_sums[c] += matrix[r2][c]

        # Find max subarray sum in col_sums no larger than k
        prefix_sum = 0
        sorted_prefix_sums = SortedList([0])

        for s in col_sums:
          prefix_sum += s
          # We want prefix_sum - prev >= target => prev >= prefix_sum - k
          target = prefix_sum - k
          idx = sorted_prefix_sums.bisect_left(target)
          if idx < len(sorted_prefix_sums):
            max_sum = max(max_sum, prefix_sum - sorted_prefix_sums[idx])
          sorted_prefix_sums.add(prefix_sum)

          if max_sum == k:
            return k

    return max_sum

```
